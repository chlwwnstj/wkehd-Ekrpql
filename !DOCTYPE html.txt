  <!-- 카카오맵 SDK (autoload=false: 우리가 원하는 때 직접 로딩) -->
  <script src="//dapi.kakao.com/v2/maps/sdk.js?appkey=654464dfc5d446f2196b6b0a69e4ba29&libraries=services&autoload=false"></script>

  <script>
    // 지도, 장소 검색 객체, 마커 목록을 저장할 변수
    let map;
    let placesService;
    let markerList = [];

    // 카카오맵 SDK가 완전히 로드된 뒤에 실행 (로딩 오류 방지)
    kakao.maps.load(function () {
      // 위치 정보를 못 가져올 때 사용할 기본 위치 (서울 시청)
      const defaultPosition = new kakao.maps.LatLng(37.5665, 126.9780);

      // 지도 생성 (확대 레벨은 5)
      const mapOptions = {
        center: defaultPosition,
        level: 5
      };
      map = new kakao.maps.Map(document.getElementById("map"), mapOptions);

      // 장소 검색 객체 생성
      placesService = new kakao.maps.services.Places();

      // 현재 위치를 가져와서 지도 중심으로 설정
      moveToCurrentPosition();

      // 이벤트 연결
      document.getElementById("searchButton").addEventListener("click", searchPlaces);
      document.getElementById("keywordInput").addEventListener("keydown", function (event) {
        if (event.key === "Enter") {
          searchPlaces();
        }
      });
    });

    // 현재 위치로 지도 이동하기
    function moveToCurrentPosition() {
      // 브라우저가 위치 기능을 지원하는지 확인
      if (!navigator.geolocation) {
        alert("이 브라우저는 위치 기능을 지원하지 않아요.");
        return;
      }

      // 위치 권한을 요청하고 현재 위치 가져오기
      navigator.geolocation.getCurrentPosition(
        function (position) {
          const currentPosition = new kakao.maps.LatLng(
            position.coords.latitude,
            position.coords.longitude
          );
          map.setCenter(currentPosition);
        },
        function (error) {
          // 권한을 거부했거나 오류가 났을 때는 기본 위치를 그대로 사용
          console.log("위치 정보를 가져오지 못했어요:", error.message);
        }
      );
    }

    // 키워드로 장소 검색하기
    function searchPlaces() {
      const keyword = document.getElementById("keywordInput").value.trim();

      if (keyword === "") {
        alert("검색어를 입력해 주세요!");
        return;
      }

      // 현재 지도 중심 근처를 우선으로 검색
      placesService.keywordSearch(keyword, handleSearchResult, {
        location: map.getCenter()
      });
    }

    // 검색 결과를 받은 뒤 실행되는 함수
    function handleSearchResult(data, status) {
      const resultList = document.getElementById("resultList");
      resultList.innerHTML = "";
      clearMarkers();

      if (status === kakao.maps.services.Status.ZERO_RESULT) {
        resultList.innerHTML = '<li class="message">검색 결과가 없어요.</li>';
        return;
      }

      if (status === kakao.maps.services.Status.ERROR) {
        resultList.innerHTML = '<li class="message">검색 중 오류가 발생했어요.</li>';
        return;
      }

      // 모든 결과가 한 화면에 보이도록 범위를 계산할 객체
      const bounds = new kakao.maps.LatLngBounds();

      data.forEach(function (place) {
        const position = new kakao.maps.LatLng(place.y, place.x);

        // 마커 만들기
        const marker = new kakao.maps.Marker({
          map: map,
          position: position
        });
        markerList.push(marker);
        bounds.extend(position);

        // 리스트 항목 만들기
        const listItem = document.createElement("li");
        listItem.innerHTML =
          '<div class="place-name">' + place.place_name + "</div>" +
          '<div class="place-address">' + (place.road_address_name || place.address_name) + "</div>";

        // 리스트를 클릭하면 해당 장소로 이동
        listItem.addEventListener("click", function () {
          map.setCenter(position);
          map.setLevel(3);
        });

        resultList.appendChild(listItem);
      });

      // 검색된 장소들이 모두 보이도록 지도 범위 조정
      map.setBounds(bounds);
    }

    // 기존 마커 모두 지우기
    function clearMarkers() {
      markerList.forEach(function (marker) {
        marker.setMap(null);
      });
      markerList = [];
    }
  </script>
